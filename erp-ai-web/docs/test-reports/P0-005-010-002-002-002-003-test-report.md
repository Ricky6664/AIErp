## useMutation API 设计

```ts
// 创建
const { state, mutate, reset, isSuccess } = useMutation({
  create: (data) => fieldConfigApi.createFieldConfig(data),
  update: (data) => fieldConfigApi.updateFieldConfig(data.id, data),
  onSuccess: (result) => emit('saved', result),
  autoDetectMode: true
})

// 保存流程
const result = await mutate(formData)
if (result) {
  ElMessage.success('保存成功')
} else {
  ElMessage.error(state.error?.message || '保存失败')
}
```
