### 5.1 交互逻辑

```typescript
const searchForm = reactive({
  applicantId: null as number | null,
  leaveType: null as string | null,
  dateRange: [] as string[],
  status: null as string | null
})

// 搜索：重置pageNum=1
const handleSearch = () => {
  const params = {
    applicantId: searchForm.applicantId || undefined,
    leaveType: searchForm.leaveType || undefined,
    startDate: searchForm.dateRange?.[0] || undefined,
    endDate: searchForm.dateRange?.[1] || undefined,
    status: searchForm.status || undefined,
    pageNum: 1
  }
  emit('search', params)
  syncUrlParams(params)
}

// 重置
const handleReset = () => {
  Object.assign(searchForm, {
    applicantId: null,
    leaveType: null,
    dateRange: [],
    status: null
  })
  handleSearch()
}

// URL参数同步
const syncUrlParams = (params: Record<string, any>) => {
  const query: Record<string, string> = {}
  if (params.applicantId) query.applicantId = String(params.applicantId)
  if (params.leaveType) query.leaveType = params.leaveType
  if (params.startDate) query.startDate = params.startDate
  if (params.endDate) query.endDate = params.endDate
  if (params.status) query.status = params.status
  router.replace({ query })
}

// 页面加载时从URL恢复搜索条件
onMounted(() => {
  const { applicantId, leaveType, startDate, endDate, status } = route.query
  if (applicantId) searchForm.applicantId = Number(applicantId)
  if (leaveType) searchForm.leaveType = leaveType as string
  if (startDate && endDate) searchForm.dateRange = [startDate as string, endDate as string]
  if (status) searchForm.status = status as string
  handleSearch()
})
```

### 5.2 各字段触发方式

| 字段 | 触发方式 | 说明 |
|------|---------|------|
| applicantId | @change即触发 | UserSelector选中即搜索 |
| leaveType | @change即触发 | 下拉选择变化即搜索 |
| dateRange | @change即触发 | 日期范围选择后自动搜索 |
| status | @change即触发 | 下拉选择变化即搜索 |

### 5.3 URL参数同步

| URL参数 | 对应表单字段 | 类型转换 |
|---------|------------|---------|
| applicantId | searchForm.applicantId | Number() |
| leaveType | searchForm.leaveType | string |
| startDate | dateRange[0] | string |
| endDate | dateRange[1] | string |
| status | searchForm.status | string |
