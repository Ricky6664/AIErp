```java
@RestController
@Tag(name = "期末处理-Period管理", description = "Period相关接口")
public class PeriodController {

    @PostMapping("/create")
    @Operation(summary = "创建Period", description = "创建新的Period记录")
    public Result<PeriodVO> create(
        @RequestBody @Valid @Parameter(description = "创建参数") PeriodCreateDTO dto) {
        ...
    }
}

@Data
public class PeriodCreateDTO {
    @Schema(description = "编号", example = "EXP202605260001", required = true)
    @NotBlank private String code;
}
```
