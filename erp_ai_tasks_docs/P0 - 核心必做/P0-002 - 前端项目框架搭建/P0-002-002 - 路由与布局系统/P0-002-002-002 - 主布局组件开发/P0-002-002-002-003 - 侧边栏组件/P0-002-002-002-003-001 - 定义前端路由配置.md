### 5.1 配置结构

**文件：`erp-web/src/layouts/components/Sidebar/types.ts`**

```typescript
import type { RouteRecordRaw } from 'vue-router'

/** 侧边栏组件Props */
export interface SidebarProps {
  isCollapsed: boolean  // 是否折叠(来自layoutStore)
}

/** 菜单项数据结构(供SidebarItem使用) */
export interface MenuItemData {
  path: string           // 路由路径
  title: string          // 菜单标题
  titleI18n?: string     // i18n key
  icon?: string          // 图标名称
  children?: MenuItemData[]  // 子菜单
  hideMenu?: boolean     // 是否隐藏
  meta?: Record<string, any>  // 路由meta信息
}

/** 将RouteRecordRaw转换为MenuItemData */
export function routeToMenuItem(route: RouteRecordRaw): MenuItemData {
  return {
    path: route.path,
    title: (route.meta?.title as string) || '',
    titleI18n: route.meta?.titleI18n as string | undefined,
    icon: route.meta?.icon as string | undefined,
    hideMenu: route.meta?.hideMenu as boolean | undefined,
    children: route.children?.map(routeToMenuItem),
    meta: route.meta
  }
}
```

### 5.2 关键配置项

| 配置项 | 类型 | 用途 |
|-------|------|------|
| SidebarProps.isCollapsed | boolean | 控制el-menu的collapse属性 |
| MenuItemData.path | string | el-menu-item的index属性 |
| MenuItemData.children | MenuItemData[] | 判断渲染el-sub-menu还是el-menu-item |
| MenuItemData.hideMenu | boolean | 过滤不在菜单显示的项 |

### 5.3 与路由守卫的集成

1. permissionStore.routes包含所有动态路由，侧边栏从中过滤hideMenu=false的项
2. layoutStore.isCollapsed通过props传递给Sidebar组件
3. routeToMenuItem函数将Vue Router的路由对象转换为菜单组件所需的扁平结构

## 八、易错警示

> ⚠️ MenuItemData不要直接使用RouteRecordRaw，因为路由对象包含component等不需要在菜单中暴露的属性

> ⚠️ routeToMenuItem递归处理children时注意空数组和undefined的区别

> ⚠️ hideMenu过滤应在渲染层(v-if)而非数据层，保持原始路由数据完整
