### 5.1 配置结构

**文件：`erp-web/src/layouts/components/Sidebar/menuConfig.ts`**

```typescript
import type { RouteRecordRaw } from 'vue-router'
import type { MenuItemData } from './types'

/**
 * 从路由列表中过滤出菜单数据
 * 排除hideMenu=true的路由和无可见子路由的目录
 */
export function filterMenuRoutes(routes: RouteRecordRaw[]): MenuItemData[] {
  return routes
    .filter(route => !route.meta?.hideMenu)
    .map(route => {
      const item: MenuItemData = {
        path: route.path,
        title: (route.meta?.title as string) || '',
        titleI18n: route.meta?.titleI18n as string,
        icon: route.meta?.icon as string,
        hideMenu: false,
        meta: route.meta,
        children: route.children ? filterMenuRoutes(route.children) : undefined
      }
      return item
    })
    .filter(item => {
      // 过滤掉没有可见子路由的目录
      if (item.children && item.children.length === 0) return false
      return true
    })
}

/** 默认展开的菜单路径列表 */
export const DEFAULT_OPEN_MENUS: string[] = [
  '/home'
]

/** 侧边栏Logo配置 */
export const SIDEBAR_LOGO = {
  collapsed: '/logo-mini.svg',
  expanded: '/logo-full.svg',
  title: 'ERP管理系统',
  link: '/home'
}
```

### 5.2 关键配置项

| 配置项 | 类型 | 用途 |
|-------|------|------|
| filterMenuRoutes | Function | 从路由列表过滤菜单项 |
| DEFAULT_OPEN_MENUS | string[] | el-menu default-openeds属性 |
| SIDEBAR_LOGO | Object | Logo图片路径和标题配置 |

### 5.3 与路由守卫的集成

1. Sidebar组件通过`permissionStore.routes`获取完整路由列表
2. 调用`filterMenuRoutes()`过滤出可见菜单项
3. 过滤结果通过computed缓存，路由变化时自动更新
4. DEFAULT_OPEN_MENUS配置初始展开状态，用户操作后由el-menu内部管理

## 八、易错警示

> ⚠️ filterMenuRoutes是纯函数，不要修改原始RouteRecordRaw对象

> ⚠️ 过滤空子路由目录时要注意：仅当所有子项都被过滤时才隐藏父目录

> ⚠️ SIDEBAR_LOGO的图片路径必须是public目录下的静态资源，不能使用@/assets路径
