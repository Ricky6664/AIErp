### 5.1 配置结构

**文件：`erp-web/src/api/modules/menu.ts` — 菜单API**

```typescript
import request from '@/utils/request'
import type { MenuItem, MenuResponse } from '@/api/types/menu'

/** 获取用户菜单和权限信息 */
export function getMenuList(): Promise<MenuResponse> {
  return request.get('/api/system/user/getInfo')
}

/** 扁平化菜单树(用于权限标识查找) */
export function flattenMenuTree(menus: MenuItem[]): MenuItem[] {
  const result: MenuItem[] = []
  function traverse(items: MenuItem[]) {
    items.forEach(item => {
      result.push(item)
      if (item.children?.length) {
        traverse(item.children)
      }
    })
  }
  traverse(menus)
  return result
}

/** 从菜单树中查找权限标识列表 */
export function extractPermissions(menus: MenuItem[]): string[] {
  const flat = flattenMenuTree(menus)
  return flat
    .filter(item => item.permissions)
    .map(item => item.permissions!)
}
```

### 5.2 关键配置项

| 函数/类型 | 用途 |
|----------|------|
| getMenuList() | 调用后端getInfo接口获取用户菜单树和权限列表 |
| flattenMenuTree() | 将嵌套菜单树扁平化为一维数组 |
| extractPermissions() | 从菜单树中提取所有权限标识 |
| MenuResponse | API响应类型，包含menus和permissions |

### 5.3 与路由守卫的集成

1. 路由守卫中调用`userStore.getUserInfo()` → 内部调用`getMenuList()`
2. 获取的menus传给`permissionStore.generateRoutes()`生成动态路由
3. 获取的permissions存入permissionStore用于权限判断
4. flattenMenuTree用于权限标识的快速查找和校验

## 八、易错警示

> ⚠️ flattenMenuTree使用递归遍历，菜单层级过深(>10)时注意栈溢出风险

> ⚠️ extractPermissions过滤掉permissions为空的项，避免空字符串进入权限列表

> ⚠️ getMenuList请求必须携带token(Axios拦截器自动添加)，否则后端返回401
