### 5.1 文件结构与组件设计

**文件：`erp-web/src/utils/permission.ts` — 权限过滤工具函数**

```typescript
import type { RouteRecordRaw } from 'vue-router'

/**
 * 检查用户是否有权限访问路由
 * @param routePermissions 路由需要的权限列表
 * @param userPermissions 用户拥有的权限列表
 */
export function hasPermission(
  routePermissions: string[] | undefined,
  userPermissions: string[]
): boolean {
  // 无权限要求的路由，所有人可访问
  if (!routePermissions || routePermissions.length === 0) return true
  // 用户拥有其中任一权限即可
  return routePermissions.some(p => userPermissions.includes(p))
}

/**
 * 根据用户权限过滤路由树(纯函数，不修改原始数据)
 * 父菜单所有子项都无权限时，父菜单也隐藏
 */
export function filterRoutesByPermission(
  routes: RouteRecordRaw[],
  userPermissions: string[]
): RouteRecordRaw[] {
  return routes.reduce<RouteRecordRaw[]>((acc, route) => {
    const routePerms = route.meta?.permissions as string[] | undefined

    // 检查当前路由权限
    if (!hasPermission(routePerms, userPermissions)) return acc

    // 递归过滤子路由
    const filteredRoute = { ...route }
    if (route.children?.length) {
      filteredRoute.children = filterRoutesByPermission(
        route.children,
        userPermissions
      )
      // 如果所有子项都被过滤，父路由也不显示
      if (filteredRoute.children.length === 0) return acc
    }

    acc.push(filteredRoute)
    return acc
  }, [])
}
```

**Sidebar组件中使用：**
```typescript
import { usePermissionStore } from '@/stores/modules/permission'

const permissionStore = usePermissionStore()

const filteredMenus = computed(() => {
  return filterRoutesByPermission(
    permissionStore.routes,
    permissionStore.permissions
  )
})
```

### 5.2 核心实现逻辑

1. **hasPermission**：some()匹配，用户有任一权限即可访问；无权限要求的路由默认放行
2. **filterRoutesByPermission**：递归过滤，使用reduce创建新数组不修改原始数据
3. **父菜单联动**：子项全被过滤时父菜单也隐藏(if children.length === 0)
4. **computed缓存**：filteredMenus在permissions变化时自动重新计算
5. **仅影响UI**：权限过滤只影响侧边栏渲染，不影响路由注册(安全由后端保障)

### 5.3 交互逻辑

1. 用户登录 → permissionStore.permissions赋值 → filteredMenus computed计算
2. 渲染侧边栏 → 遍历filteredMenus → 无权限菜单不出现在DOM中
3. 权限变更(管理员修改角色) → 重新登录 → permissions更新 → 菜单自动更新
4. 直接输入URL访问无权限路由 → 路由守卫拦截 → 跳转/no-permission

## 八、易错警示

> ⚠️ 权限过滤只影响UI显示，不影响路由注册，安全由后端守卫保障

> ⚠️ filterRoutesByPermission必须使用纯函数，{...route}创建浅拷贝，不修改原始路由对象

> ⚠️ 父菜单联动：所有子项被过滤时父菜单也要隐藏，避免显示空的展开菜单

> ⚠️ 超级管理员权限('admin'或'*')需特殊处理，hasPermission中优先判断
