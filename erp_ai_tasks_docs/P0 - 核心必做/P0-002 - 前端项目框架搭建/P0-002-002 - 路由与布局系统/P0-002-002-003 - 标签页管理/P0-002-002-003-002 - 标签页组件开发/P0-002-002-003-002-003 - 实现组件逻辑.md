### 5.1 文件结构与组件设计

文件路径：`erp-web/src/layouts/components/TabNav/index.vue`（script逻辑部分）

```vue
<script setup lang="ts">
import { ref, computed, watch, nextTick } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { Close } from '@element-plus/icons-vue'
import ContextMenu from './ContextMenu.vue'
import { useTagsViewStore } from '@/stores/modules/tagsView'
import type { TagView } from '@/stores/modules/tagsView'

defineOptions({ name: 'TabNav' })

const route = useRoute()
const router = useRouter()
const tagsViewStore = useTagsViewStore()

const scrollRef = ref<HTMLElement>()
const contextMenuRef = ref<InstanceType<typeof ContextMenu>>()
const selectedTag = ref<TagView>()

// 从store获取标签列表
const views = computed(() => tagsViewStore.visitedViews)
const activePath = computed(() => route.fullPath)

// 监听路由变化，自动添加标签
watch(
  () => route.fullPath,
  () => {
    // 排除不需要添加标签的路由
    if (route.meta?.hideTab) return
    if (!route.name) return
    tagsViewStore.addView(route)
    // 自动滚动到激活标签
    nextTick(() => scrollToActiveTag())
  },
  { immediate: true }
)

// 点击标签切换路由
function handleSelect(path: string) {
  if (path !== route.fullPath) {
    router.push(path)
  }
}

// 关闭标签
function handleClose(tag: TagView) {
  tagsViewStore.closeSelectedTag(tag)
}

// 右键菜单
function handleContextmenu(event: MouseEvent, tag: TagView) {
  selectedTag.value = tag
  contextMenuRef.value?.open(event)
}

// 滚动到激活标签
function scrollToActiveTag() {
  if (!scrollRef.value) return
  const activeEl = scrollRef.value.querySelector('.tab-nav__item--active')
  if (activeEl) {
    activeEl.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'nearest' })
  }
}
</script>
```

### 5.2 核心实现逻辑

1. **watch route.fullPath**：路由变化时自动调用tagsViewStore.addView添加标签
2. **computed activePath**：从当前route.fullPath派生，驱动标签高亮
3. **hideTab过滤**：meta.hideTab=true的路由不添加标签(如login/404)
4. **scrollToActiveTag**：标签过多时自动滚动scroll容器使激活标签可见
5. **ContextMenu引用**：通过ref获取实例，调用open方法打开右键菜单

### 5.3 交互逻辑

1. 路由变化 → watch触发 → addView → 标签列表更新 → scrollToActiveTag
2. 点击标签 → handleSelect → router.push → route变化 → watch再次触发
3. 右键标签 → handleContextmenu → 设置selectedTag → contextMenuRef.open
4. 关闭标签 → handleClose → closeSelectedTag → 如果是当前标签则导航到相邻标签

## 八、易错警示

> ⚠️ watch的immediate:true确保首次加载时也添加标签，否则首页标签不显示

> ⚠️ hideTab判断使用route.meta?.hideTab，登录页/404等不添加标签

> ⚠️ scrollToActiveTag必须在nextTick后执行，确保DOM已渲染

> ⚠️ handleSelect中判断path !== route.fullPath避免重复导航触发守卫
