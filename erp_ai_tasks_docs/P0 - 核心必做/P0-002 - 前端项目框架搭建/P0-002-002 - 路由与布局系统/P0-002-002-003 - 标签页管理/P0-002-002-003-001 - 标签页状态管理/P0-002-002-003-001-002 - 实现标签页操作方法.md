### 5.1 文件结构与组件设计

在`erp-web/src/stores/modules/tagsView.ts`中追加以下actions:

```typescript
import router from '@/router'
import { nextTick } from 'vue'

/** 关闭标签并导航到下一个标签 */
function closeSelectedTag(view: TagView) {
  delView(view)
  // 如果关闭的是当前激活标签，需要导航到相邻标签
  if (isActiveView(view)) {
    const latestView = visitedViews.value.slice(-1)[0]
    if (latestView) {
      router.push(latestView.fullPath)
    } else {
      router.push('/home')
    }
  }
}

/** 刷新当前页面(通过exclude+include实现) */
async function refreshSelectedPage(route: RouteLocationNormalized) {
  const name = route.name as string
  // 1. 从缓存列表移除
  delCachedView({ name } as TagView)
  // 2. 导航到redirect中转页
  await router.replace({ path: '/redirect' + route.fullPath })
  // 3. nextTick后重新加入缓存列表
  await nextTick()
  addCachedView(route)
}

/** 关闭左侧标签 */
function closeLeftTags(currentView: TagView) {
  const currentIndex = visitedViews.value.findIndex(
    v => v.fullPath === currentView.fullPath
  )
  visitedViews.value = visitedViews.value.filter((v, i) =>
    i >= currentIndex || v.affix
  )
  syncCachedViews()
}

/** 关闭右侧标签 */
function closeRightTags(currentView: TagView) {
  const currentIndex = visitedViews.value.findIndex(
    v => v.fullPath === currentView.fullPath
  )
  visitedViews.value = visitedViews.value.filter((v, i) =>
    i <= currentIndex || v.affix
  )
  syncCachedViews()
}

/** 同步cachedViews与visitedViews */
function syncCachedViews() {
  cachedViews.value = visitedViews.value
    .filter(v => v.keepAlive)
    .map(v => v.name as string)
    .filter(Boolean)
}
```

### 5.2 核心实现逻辑

1. **closeSelectedTag**：关闭标签后查找最后一个visitedView作为下一个导航目标
2. **refreshSelectedPage**：先exclude → 导航到redirect中转页 → nextTick后重新include → 组件重建
3. **closeLeftTags/closeRightTags**：基于当前标签索引位置过滤，保留affix标签
4. **syncCachedViews**：批量操作后统一同步缓存列表

### 5.3 交互逻辑

1. 点击标签关闭按钮 → closeSelectedTag → 标签移除 → 如果是当前标签则导航到相邻标签
2. 右键"刷新" → refreshSelectedPage → 组件销毁重建 → 数据重新加载
3. 右键"关闭左侧" → closeLeftTags → 左侧标签批量移除 → 缓存同步
4. 右键"关闭右侧" → closeRightTags → 右侧标签批量移除 → 缓存同步

## 八、易错警示

> ⚠️ closeSelectedTag关闭当前标签时必须计算正确的下一个路由，避免跳到错误页面

> ⚠️ refreshSelectedPage使用redirect中转页而非location.reload()，避免丢失所有状态

> ⚠️ 批量操作(closeLeft/closeRight/Other/All)后必须调用syncCachedViews同步缓存列表

> ⚠️ affix标签在所有关闭操作中都要保留，通过filter条件`v.affix`确保
