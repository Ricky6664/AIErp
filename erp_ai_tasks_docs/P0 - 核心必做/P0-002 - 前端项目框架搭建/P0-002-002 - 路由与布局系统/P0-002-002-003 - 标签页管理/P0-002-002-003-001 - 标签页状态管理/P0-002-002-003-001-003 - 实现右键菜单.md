### 5.1 文件结构与组件设计

文件路径：`erp-web/src/layouts/components/TabNav/ContextMenu.vue`

```vue
<template>
  <Teleport to="body">
    <ul
      v-show="visible"
      class="tab-context-menu"
      :style="{ left: position.x + 'px', top: position.y + 'px' }"
    >
      <li @click="handleAction('refresh')">
        <el-icon><Refresh /></el-icon> 刷新当前页
      </li>
      <li
        :class="{ 'is-disabled': selectedTag?.affix }"
        @click="handleAction('close')"
      >
        <el-icon><Close /></el-icon> 关闭当前
      </li>
      <li @click="handleAction('closeOther')">
        <el-icon><Switch /></el-icon> 关闭其他
      </li>
      <li @click="handleAction('closeLeft')">
        <el-icon><Back /></el-icon> 关闭左侧
      </li>
      <li @click="handleAction('closeRight')">
        <el-icon><Right /></el-icon> 关闭右侧
      </li>
      <li @click="handleAction('closeAll')">
        <el-icon><Minus /></el-icon> 关闭全部
      </li>
    </ul>
  </Teleport>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue'
import { Refresh, Close, Switch, Back, Right, Minus } from '@element-plus/icons-vue'
import { useTagsViewStore } from '@/stores/modules/tagsView'
import type { TagView } from '@/stores/modules/tagsView'

defineOptions({ name: 'TabContextMenu' })

interface Props {
  selectedTag?: TagView
}
const props = defineProps<Props>()

const visible = ref(false)
const position = ref({ x: 0, y: 0 })
const tagsViewStore = useTagsViewStore()

function open(event: MouseEvent) {
  event.preventDefault()
  position.value = { x: event.clientX, y: event.clientY }
  visible.value = true
}

function close() {
  visible.value = false
}

function handleAction(action: string) {
  if (!props.selectedTag) return
  if (action === 'close' && props.selectedTag.affix) return  // affix标签不可关闭

  switch (action) {
    case 'refresh':
      tagsViewStore.refreshSelectedPage(/* currentRoute */)
      break
    case 'close':
      tagsViewStore.closeSelectedTag(props.selectedTag)
      break
    case 'closeOther':
      tagsViewStore.delOtherViews(props.selectedTag)
      break
    case 'closeLeft':
      tagsViewStore.closeLeftTags(props.selectedTag)
      break
    case 'closeRight':
      tagsViewStore.closeRightTags(props.selectedTag)
      break
    case 'closeAll':
      tagsViewStore.delAllViews()
      break
  }
  close()
}

// 点击菜单外区域关闭
function handleClickOutside() { close() }

defineExpose({ open, close })
onMounted(() => document.addEventListener('click', handleClickOutside))
onUnmounted(() => document.removeEventListener('click', handleClickOutside))
</script>
```

### 5.2 核心实现逻辑

1. **Teleport to body**：菜单挂载到body层级，避免被父元素overflow裁剪
2. **position定位**：使用鼠标clientX/clientY定位菜单
3. **affix禁用**：affix标签的"关闭当前"选项添加is-disabled样式
4. **click outside关闭**：document click事件监听，点击任意位置关闭菜单
5. **defineExpose**：暴露open/close方法供TabNav组件调用

### 5.3 交互逻辑

1. 标签上右键 → contextmenu事件 → ContextMenu.open(event) → 菜单显示在鼠标位置
2. 选择菜单项 → handleAction → 调用tagsViewStore对应方法 → 菜单关闭
3. 点击菜单外区域 → document click → ContextMenu.close()
4. affix标签右键 → "关闭当前"项灰色不可点击

## 八、易错警示

> ⚠️ 右键菜单用Teleport挂载到body，避免被父元素overflow裁剪导致菜单不可见

> ⚠️ contextmenu事件必须调用event.preventDefault()阻止浏览器默认右键菜单

> ⚠️ document click监听在onUnmounted时必须移除，避免内存泄漏

> ⚠️ 菜单位置需考虑边界情况：鼠标靠近屏幕右侧时菜单可能超出视口，需做边界检测
