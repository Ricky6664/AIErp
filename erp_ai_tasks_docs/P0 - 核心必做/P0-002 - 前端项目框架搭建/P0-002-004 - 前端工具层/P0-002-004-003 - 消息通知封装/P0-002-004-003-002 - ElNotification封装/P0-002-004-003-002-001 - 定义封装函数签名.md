### 5.1 函数签名与类型定义

```typescript
// erp-web/src/utils/notification.ts

import type { VNode } from 'vue';
import type { NotificationProps } from 'element-plus';

/** 通知内容类型 */
type NotificationContent = string | VNode;

/** 通知类型 */
type NotificationType = 'success' | 'warning' | 'info' | 'error';

/** 通知位置 */
type NotificationPosition = 'top-right' | 'top-left' | 'bottom-right' | 'bottom-left';

/** 通知配置选项 */
interface NotifyOptions {
  /** 通知标题 */
  title: string;
  /** 通知内容 */
  message: NotificationContent;
  /** 通知类型，默认'info' */
  type?: NotificationType;
  /** 显示时长(ms)，默认4500，0表示不自动关闭 */
  duration?: number;
  /** 通知位置，默认'top-right' */
  position?: NotificationPosition;
  /** 是否显示关闭按钮，默认true */
  showClose?: boolean;
  /** 去重标识（相同name的通知不重复弹出） */
  name?: string;
  /** 偏移量(px)，默认0 */
  offset?: number;
}

/**
 * 显示通知
 */
export function notify(options: NotifyOptions): void;

/**
 * 快捷方法：成功通知
 */
export function notifySuccess(title: string, message: NotificationContent): void;

/**
 * 快捷方法：错误通知
 */
export function notifyError(title: string, message: NotificationContent): void;

/**
 * 快捷方法：警告通知
 */
export function notifyWarning(title: string, message: NotificationContent): void;

/**
 * 关闭所有通知
 */
export function clearAllNotifications(): void;
```

### 5.2 实现逻辑

1. **参数设计**：使用NotifyOptions对象，title和message为必填，其余为可选配置
2. **默认配置**：
   - duration: 4500ms（比Message长，适合长时间展示的通知）
   - position: 'top-right'（右上角）
   - showClose: true
3. **name去重**：name属性是ElNotification去重的关键，相同name的通知不会重复弹出
4. **快捷方法**：notifySuccess/notifyError/notifyWarning内部调用notify并固定type参数

**与ElMessage的区别**：
- ElMessage：短暂提示，居中显示，适合操作反馈（成功/失败）
- ElNotification：持久通知，角落显示，适合需要用户关注的信息（系统通知、审批提醒）

### 5.3 使用场景

1. **系统通知**：`notify({ title: '审批提醒', message: '您有一条待审批的采购单', type: 'info' })`
2. **后台任务完成**：`notifySuccess('导出完成', '报表已生成，点击下载')`
3. **错误通知**：`notifyError('同步失败', '数据同步异常，请联系管理员')`
4. **自定义位置**：`notify({ title: '提示', message: '...', position: 'bottom-right' })`
