### 5.1 函数签名与类型定义

```typescript
// erp-web/src/utils/message.ts

import type { VNode } from 'vue';
import type { ElMessageBoxOptions } from 'element-plus';

/** 消息内容类型：支持纯文本和VNode */
type MessageContent = string | VNode;

/** 消息类型 */
type MessageType = 'success' | 'warning' | 'info' | 'error';

/** 消息配置选项 */
interface MessageOptions {
  /** 消息内容 */
  message: MessageContent;
  /** 消息类型，默认'info' */
  type?: MessageType;
  /** 显示时长(ms)，默认3000 */
  duration?: number;
  /** 是否显示关闭按钮，默认true */
  showClose?: boolean;
  /** 消息分组标识（相同grouping的消息不重复弹出） */
  grouping?: boolean;
}

/**
 * 显示成功消息
 */
export function showSuccess(message: MessageContent, options?: Partial<MessageOptions>): void;

/**
 * 显示错误消息
 */
export function showError(message: MessageContent, options?: Partial<MessageOptions>): void;

/**
 * 显示警告消息
 */
export function showWarning(message: MessageContent, options?: Partial<MessageOptions>): void;

/**
 * 显示信息消息
 */
export function showInfo(message: MessageContent, options?: Partial<MessageOptions>): void;

/**
 * 确认弹窗（Promise化）
 * @returns Promise<boolean> 确认返回true，取消reject
 */
export function confirm(message: MessageContent, title?: string, options?: ElMessageBoxOptions): Promise<boolean>;
```

### 5.2 实现逻辑

1. **参数设计**：使用Options对象而非位置参数，方便扩展和可选配置
2. **默认配置**：duration=3000ms、showClose=true、grouping=true（相同内容不重复弹出）
3. **类型重载**：message参数支持string和VNode两种类型，TypeScript通过联合类型声明
4. **confirm返回Promise**：确认按钮resolve(true)，取消按钮reject('cancel')，不要用catch吞掉

**统一图标配置**：
- success → CircleCheck图标
- error → CircleClose图标
- warning → Warning图标
- info → InfoFilled图标

### 5.3 使用场景

1. **操作反馈**：`showSuccess('保存成功')` 表单提交成功后提示
2. **错误提示**：`showError('网络请求失败')` API异常时提示
3. **删除确认**：`const ok = await confirm('确认删除该记录？', '提示'); if(ok) { deleteRecord() }`
4. **VNode内容**：`showInfo(h('div', {}, [h('b', '标题'), h('p', '详情')]))`
