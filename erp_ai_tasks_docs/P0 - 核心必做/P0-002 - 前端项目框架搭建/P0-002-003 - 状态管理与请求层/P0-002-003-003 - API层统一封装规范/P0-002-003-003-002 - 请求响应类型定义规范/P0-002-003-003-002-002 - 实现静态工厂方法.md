### 5.1 工具函数定义

在`erp-web/src/utils/apiHelper.ts`中实现运行时工具函数：

```typescript
import type { ApiResponse } from '@/types/api'

/** 判断API响应是否成功（code === 0） */
export function isSuccess<T>(response: ApiResponse<T>): boolean {
  return response.code === 0
}

/** 获取API响应的错误信息 */
export function getErrorMessage(response: ApiResponse): string {
  return response.message || '未知错误'
}

/** 类型守卫：确认响应为成功状态 */
export function assertSuccess<T>(response: ApiResponse<T>): asserts response is ApiResponse<T> & { data: T } {
  if (response.code !== 0) {
    throw new Error(response.message || '请求失败')
  }
}
```

### 5.2 类型声明

在`erp-web/src/types/api.d.ts`中声明工具函数类型签名：
```typescript
export function isSuccess<T>(response: ApiResponse<T>): boolean
export function getErrorMessage(response: ApiResponse): string
export function assertSuccess<T>(response: ApiResponse<T>): asserts response is ApiResponse<T> & { data: T }
```

### 5.3 使用场景

```typescript
const result = await apiClient.get<ApiResponse<UserVO>>('/api/user/1')
if (isSuccess(result.data)) {
  console.log(result.data.data)  // TypeScript推断为UserVO
}
```

## 八、易错警示

> ⚠️ 工具函数如果不需要运行时使用（如纯类型操作），只在类型层声明即可，不需要运行时实现

> ⚠️ isSuccess判断`code === 0`，不要使用truthy判断（code为0时falsy）

> ⚠️ assertSuccess使用`asserts response is ...`语法（TypeScript 3.7+），确保tsconfig中target不低于ES2020

> ⚠️ 工具函数使用named export，不使用default export
