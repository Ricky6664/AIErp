### 5.1 静态路由配置

在`erp-web/src/router/index.ts`中定义`constantRoutes: RouteRecordRaw[]`：
```typescript
export const constantRoutes: RouteRecordRaw[] = [
  { path: '/login', component: () => import('@/views/login/index.vue'), name: 'Login', meta: { title: '登录', hidden: true } },
  { path: '/404', component: () => import('@/views/error/404.vue'), name: 'NotFound', meta: { title: '404', hidden: true } },
  { path: '/403', component: () => import('@/views/error/403.vue'), name: 'Forbidden', meta: { title: '403', hidden: true } },
  {
    path: '/',
    component: () => import('@/layout/index.vue'),
    redirect: '/dashboard',
    children: [
      { path: 'dashboard', component: () => import('@/views/dashboard/index.vue'), name: 'Dashboard', meta: { title: '首页', icon: 'HomeFilled', affix: true } }
    ]
  }
]
```

### 5.2 路由Meta类型定义

在`erp-web/src/types/router.d.ts`中扩展RouteMeta接口：
```typescript
declare module 'vue-router' {
  interface RouteMeta {
    title: string        // 页面标题（标签页/面包屑/document.title）
    icon?: string        // Element Plus图标名（侧边栏菜单图标）
    hidden?: boolean     // 是否在菜单中隐藏
    permission?: string  // 权限标识（用于路由级权限控制）
    affix?: boolean      // 是否固定在标签栏
    keepAlive?: boolean  // 是否缓存组件
    activeMenu?: string  // 指定高亮的菜单路径（详情页场景）
  }
}
```

### 5.3 动态路由组件映射

使用`import.meta.glob`实现动态组件导入：
```typescript
// erp-web/src/router/dynamicImport.ts
const modules = import.meta.glob('@/views/**/*.vue')
export function loadComponent(component: string) {
  return modules[`/src/views/${component}.vue`]
}
```

### 5.4 与permissionStore集成

permissionStore的`generateRoutes`action接收后端菜单数据，调用`loadComponent`映射component字符串为实际组件，通过`router.addRoute()`动态注册。

## 八、易错警示

> ⚠️ import.meta.glob的路径参数必须是字符串字面量或模板字符串，不能用变量拼接完整路径

> ⚠️ 静态路由中`hidden: true`的路由不会出现在侧边栏菜单中

> ⚠️ `path: '/:pathMatch(.*)*'`通配404路由必须在动态路由全部注册完成后再addRoute，否则会提前匹配

> ⚠️ 路由name必须全局唯一，不要与后端菜单返回的name冲突
