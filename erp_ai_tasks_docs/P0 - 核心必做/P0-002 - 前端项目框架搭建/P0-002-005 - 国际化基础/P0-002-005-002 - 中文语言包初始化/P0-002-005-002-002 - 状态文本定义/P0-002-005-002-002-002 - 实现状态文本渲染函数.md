### 5.1 词条结构设计
实现`renderStatusText`工具函数，封装状态文本的i18n翻译逻辑：
```typescript
import { i18n } from '@/i18n'

// 状态类型定义
type StatusType = 'audit' | 'enable' | 'order' | 'payment'

/**
 * 渲染状态文本
 * @param statusType 状态类型（业务域），如'audit'/'order'
 * @param statusCode 状态码，如'pending'/'approved'
 * @returns 翻译后的状态文本
 */
export function renderStatusText(statusType: StatusType, statusCode: string): string {
  const key = `status.${statusType}.${statusCode}`
  const result = i18n.global.t(key)
  // vue-i18n在未找到翻译时返回Key本身，据此判断是否缺失
  if (result === key) {
    console.warn(`[i18n] Status text not found: ${key}`)
    return `[${statusType}.${statusCode}]`
  }
  return result
}
```

### 5.2 导出与聚合
- 函数放在`erp-web/src/utils/status.ts`工具文件中（不在语言包文件中）
- 使用具名导出`export function renderStatusText`，组件按需import
- 同时导出StatusType类型供组件使用：`export type { StatusType }`
- 可在表格列的formatter或template中直接调用

### 5.3 特殊处理
- 未匹配状态码时返回`[statusType.statusCode]`格式默认文本（方便开发阶段发现问题）
- 开发环境下console.warn输出缺失的状态Key，生产环境静默
- 动态拼接Key时注意TypeScript类型安全，使用StatusType联合类型约束statusType参数
- 状态Tag组件（带颜色）应调用renderStatusText获取文本，不要在组件中硬编码中文
