### 5.2 完整配置
```ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import vueJsx from '@vitejs/plugin-vue-jsx'
import AutoImport from 'unplugin-auto-import/vite'
import Components from 'unplugin-vue-components/vite'
import { ElementPlusResolver } from 'unplugin-vue-components/resolvers'
import { resolve } from 'path'

export default defineConfig({
  plugins: [
    vue(),
    vueJsx(),
    AutoImport({
      imports: ['vue', 'vue-router', 'pinia'],
      resolvers: [ElementPlusResolver()],
      dts: 'src/auto-imports.d.ts'
    }),
    Components({
      resolvers: [ElementPlusResolver()],
      dts: 'src/components.d.ts'
    })
  ],
  resolve: {
    alias: {
      '@': resolve(__dirname, 'src'),
      '@components': resolve(__dirname, 'src/components')
    }
  },
  server: {
    port: 5173,
    host: true,
    proxy: {
      '/api': {
        target: 'http://localhost:8080',
        changeOrigin: true
      }
    }
  },
  build: {
    target: 'es2015',
    outDir: 'dist',
    chunkSizeWarningLimit: 1500,
    rollupOptions: {
      output: {
        manualChunks: {
          vendor: ['axios', 'dayjs', 'lodash-es'],
          vue: ['vue', 'vue-router', 'pinia'],
          'element-plus': ['element-plus']
        }
      }
    }
  }
})
```

### 5.3 关键配置项
| 配置项 | 作用 | 注意事项 |
|--------|------|----------|
| `plugins` 数组 | Vue/AutoImport/Components/JSX 插件链 | 顺序：vue → vueJsx → AutoImport → Components |
| `resolve.alias` | 路径别名映射 | `@` → `src`，`@components` → `src/components` |
| `server.port: 5173` | 开发服务器端口 | Vite 默认端口 |
| `server.proxy` | API 代理 | `/api` 转发到后端 `localhost:8080` |
| `changeOrigin: true` | 修改请求 Host 头 | 跨域代理必须配置 |
| `build.target: es2015` | 编译目标 | 兼容 Chrome 49+/Firefox 52+ |
| `build.chunkSizeWarningLimit` | chunk 大小警告阈值 | 1500KB，超过则终端警告 |
| `manualChunks` | 分包策略 | vendor/vue/element-plus 独立 chunk |

### 5.4 环境适配
- Vite 6.x 使用 ESM 模块，`vite.config.ts` 必须用 ESM 语法
- `resolve.alias` 需要配合 `tsconfig.json` 的 `paths` 配置（IDE 类型解析）
- `server.proxy` 仅在开发模式生效，生产环境通过 Nginx 反向代理
- `manualChunks` 配置需根据项目实际依赖调整，避免单 chunk 过大
- 部署到子路径时需配置 `base: '/sub-path/'`

## 八、易错警示

> ⚠️ `plugins` 数组中 `vue()` 必须放第一个

> ⚠️ `server.proxy` 中 `changeOrigin: true` 必须配置，否则后端收到的 Host 头不对

> ⚠️ `resolve.alias` 使用 `resolve(__dirname, 'src')` 而非字符串 `'./src'`（绝对路径更可靠）

> ⚠️ `manualChunks` 中的包名必须与 `package.json` 中一致
