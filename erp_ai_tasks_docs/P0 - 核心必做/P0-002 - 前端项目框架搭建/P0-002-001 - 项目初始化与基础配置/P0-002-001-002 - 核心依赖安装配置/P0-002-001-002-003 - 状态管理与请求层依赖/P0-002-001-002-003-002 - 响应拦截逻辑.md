### 5.2 响应拦截器
```ts
// src/utils/request.ts
import { ElMessage } from 'element-plus'
import { useUserStore } from '@/stores/modules/user'
import router from '@/router'

service.interceptors.response.use(
  (response: AxiosResponse) => {
    const { code, data, msg } = response.data

    // 业务成功：code === 0 返回 data
    if (code === 0) {
      return data
    }

    // 业务错误处理
    switch (code) {
      case 401:
        // Token 过期，触发刷新逻辑（见请求管理任务）
        handleTokenExpired()
        break
      case 403:
        ElMessage.error('权限不足')
        router.push('/403')
        break
      default:
        ElMessage.error(msg || '请求失败')
    }
    return Promise.reject(new Error(msg || 'Error'))
  },
  (error) => {
    // 网络错误/超时处理
    if (error.code === 'ECONNABORTED') {
      ElMessage.error('请求超时，请稍后重试')
    } else if (!error.response) {
      ElMessage.error('网络异常，请检查网络连接')
    } else {
      const status = error.response.status
      const messages: Record<number, string> = {
        400: '请求参数错误',
        404: '请求资源不存在',
        500: '服务器内部错误',
        502: '网关错误',
        503: '服务不可用'
      }
      ElMessage.error(messages[status] || `请求失败(${status})`)
    }
    return Promise.reject(error)
  }
)
```

### 5.3 核心实现逻辑
1. **业务响应结构**：后端统一返回 `{ code: number, data: T, msg: string }`
2. **code === 0**：业务成功，拦截器直接返回 `data` 字段（剥离外层包装）
3. **code !== 0**：业务失败，通过 `ElMessage.error` 弹出错误提示
4. **401 特殊处理**：Token 过期，触发 refreshToken 刷新逻辑
5. **HTTP 状态码**：网络层错误映射为中文提示

### 5.4 关键配置项
| 配置项 | 作用 | 注意事项 |
|--------|------|----------|
| `code === 0` 判断 | 业务成功标识 | 与后端约定一致，不可更改 |
| `response.data` 解构 | 剥离外层包装返回纯数据 | 业务代码中直接使用返回数据 |
| `ElMessage.error` | 统一错误提示 | 避免各组件自行处理错误提示 |
| `error.response.status` | HTTP 状态码 | 区分网络错误和服务器错误 |

### 5.5 环境适配
- 拦截器在 Axios 实例创建后立即注册（`src/utils/request.ts` 文件加载时）
- 401 处理需要访问 userStore（获取/刷新 Token），需确保 Pinia 已初始化
- 错误提示使用 `ElMessage` 而非 `alert`，保持 UI 一致性
- `Promise.reject(error)` 确保调用方可以继续 `.catch()` 处理

## 八、易错警示

> ⚠️ 业务 code 判断必须使用 `===`（严格相等），不能用 `==`（类型不同可能导致误判）

> ⚠️ 401 处理必须防止并发刷新：多个请求同时收到 401 时只触发一次 refreshToken

> ⚠️ `ElMessage.error` 调用前需确认 Element Plus 已按需引入 ElMessage

> ⚠️ 拦截器中 `return Promise.reject(error)` 不可遗漏，否则调用方 `.catch()` 无法捕获
