### 5.1 脚本结构总览

```bash
#!/bin/bash
set -euo pipefail

# ==================== 参数与配置 ====================
IMAGE_TAG="${1:-latest}"
DEPLOY_DIR="/opt/erp"
BACKUP_DIR="/backup/db"
LOG_FILE="/var/log/erp-deploy.log"
HEALTH_URL="http://localhost/actuator/health"
MAX_RETRIES=30
RETRY_INTERVAL=2
KEEP_BACKUPS=7
KEEP_ROLLBACK_VERSIONS=3
OPERATOR="${DEPLOY_USER:-ci-bot}"

# ==================== 主流程 ====================
log "========== 开始部署 =========="
log "镜像标签: $IMAGE_TAG"
log "操作人: $OPERATOR"

preflight_check
backup_database
pull_images
rolling_update
health_check || rollback
log "========== 部署完成 =========="
```

### 5.2 日志函数

```bash
log() {
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "[$timestamp] [$OPERATOR] $*" | tee -a "$LOG_FILE"
}

log_error() {
    log "[ERROR] $*"
}
```

- 所有输出同时打印到终端与记录到日志文件
- 日志格式：`[时间] [操作人] 消息`，便于审计
- 日志文件按天轮转（配合logrotate）

### 5.3 预检（preflight_check）

```bash
preflight_check() {
    log "执行预检..."
    # 检查Docker与Docker Compose
    command -v docker >/dev/null || { log_error "Docker未安装"; exit 1; }
    command -v docker compose >/dev/null || { log_error "Docker Compose未安装"; exit 1; }
    # 检查部署目录
    [ -d "$DEPLOY_DIR" ] || { log_error "部署目录不存在: $DEPLOY_DIR"; exit 1; }
    [ -f "$DEPLOY_DIR/docker-compose.yml" ] || { log_error "docker-compose.yml不存在"; exit 1; }
    # 检查磁盘空间（至少10GB可用）
    local free_space
    free_space=$(df "$DEPLOY_DIR" --output=avail | tail -1)
    [ "$free_space" -gt 10485760 ] || { log_error "磁盘空间不足: ${free_space}KB"; exit 1; }
    # 检查当前服务状态
    cd "$DEPLOY_DIR"
    docker compose ps >/dev/null || log "[WARN] 当前无运行中的服务（首次部署）"
    log "预检通过"
}
```

### 5.4 数据库备份（backup_database）

```bash
backup_database() {
    log "备份数据库..."
    mkdir -p "$BACKUP_DIR"
    local timestamp
    timestamp=$(date '+%Y%m%d_%H%M%S')
    local backup_file="$BACKUP_DIR/erp_${timestamp}.sql"

    # 使用docker exec执行pg_dump
    docker exec erp-postgres-1 pg_dump -U "$DB_USER" -d "$DB_NAME" > "$backup_file"

    # 压缩备份
    gzip "$backup_file"
    log "备份完成: ${backup_file}.gz"

    # 清理过期备份（保留最近N天）
    find "$BACKUP_DIR" -name "erp_*.sql.gz" -mtime +"$KEEP_BACKUPS" -delete
    log "清理超过${KEEP_BACKUPS}天的备份"
}
```

- 部署前必须备份数据库，避免更新失败时数据丢失
- 备份使用 `pg_dump` 直接输出SQL，可恢复
- 保留最近7天备份，避免磁盘占满

### 5.5 拉取镜像（pull_images）

```bash
pull_images() {
    log "拉取镜像: $IMAGE_TAG"
    cd "$DEPLOY_DIR"
    # 记录当前镜像标签用于回滚
    CURRENT_TAG=$(docker compose config | grep -oP 'image: \K.*' | head -1 | cut -d: -f2)
    echo "$CURRENT_TAG" > "$DEPLOY_DIR/.current_tag"

    # 更新.env中的TAG变量
    sed -i "s/^TAG=.*/TAG=$IMAGE_TAG/" .env

    # 拉取新镜像
    docker compose pull backend frontend
    log "镜像拉取完成"
}
```

### 5.6 滚动更新（rolling_update）

```bash
rolling_update() {
    log "执行滚动更新..."
    cd "$DEPLOY_DIR"
    # 使用--remove-orphans清理孤立容器
    # 使用--no-deps避免重启依赖服务
    docker compose up -d --remove-orphans --no-deps backend
    sleep 5
    docker compose up -d --remove-orphans --no-deps frontend
    log "滚动更新完成"
}
```

- 使用 `--no-deps` 仅更新目标服务，避免重启postgres/redis等基础设施
- 先更新backend，等5秒让backend启动后再更新frontend
- 实际场景中可配合多实例backend实现真正滚动（一个实例一个实例更新）

### 5.7 健康检查（health_check）

```bash
health_check() {
    log "执行健康检查..."
    for i in $(seq 1 $MAX_RETRIES); do
        local status
        status=$(curl -s -o /dev/null -w "%{http_code}" "$HEALTH_URL" 2>/dev/null || echo "000")
        if [ "$status" = "200" ]; then
            local body
            body=$(curl -s "$HEALTH_URL")
            if echo "$body" | grep -q '"status":"UP"'; then
                log "[OK] 应用健康 (第${i}次检查)"
                return 0
            fi
        fi
        log "[WARN] 健康检查失败 (第${i}/${MAX_RETRIES}次), status=$status"
        sleep "$RETRY_INTERVAL"
    done
    log_error "健康检查超时（${MAX_RETRIES}×${RETRY_INTERVAL}s）"
    return 1
}
```

- 循环30次，每次间隔2秒，总计最多60秒
- 不仅检查HTTP 200，还要验证响应体包含 `"status":"UP"`
- 失败时返回非0退出码，触发rollback

### 5.8 回滚机制（rollback）

```bash
rollback() {
    log_error "触发回滚..."
    cd "$DEPLOY_DIR"
    local rollback_tag
    rollback_tag=$(cat "$DEPLOY_DIR/.current_tag" 2>/dev/null || echo "latest")

    # 恢复.env中的TAG
    sed -i "s/^TAG=.*/TAG=$rollback_tag/" .env

    # 重启服务
    docker compose down backend frontend
    docker compose up -d --remove-orphans backend frontend

    log "回滚到标签: $rollback_tag"

    # 验证回滚后服务是否健康
    sleep 10
    if health_check; then
        log "[OK] 回滚成功"
    else
        log_error "回滚后服务仍不健康，需要人工介入"
    fi
    exit 1
}
```

- 回滚到部署前的镜像标签（保存在 `.current_tag` 文件中）
- 保留最近3个版本镜像，避免回滚时镜像不存在
- 回滚后仍要健康检查，若回滚也失败则记录日志并提示人工介入

### 5.9 镜像保留策略

```bash
prune_images() {
    log "清理旧镜像..."
    # 保留最近N个版本，删除其他
    docker images erp-backend --format "{{.Tag}}" | \
        grep -E '^[a-f0-9]{7,40}$' | \
        tail -n +$((KEEP_ROLLBACK_VERSIONS + 1)) | \
        xargs -r -I {} docker rmi "erp-backend:{}"
    docker system prune -f
}
```

### 5.10 完整脚本入口

```bash
main() {
    trap 'log_error "部署中断"; exit 1' INT TERM

    preflight_check
    backup_database
    pull_images
    rolling_update

    if health_check; then
        prune_images
        log "[OK] 部署成功: $IMAGE_TAG"
    else
        rollback
    fi
}

main "$@"
```

## 八、易错警示

> ⚠️ 脚本顶部必须使用 `set -euo pipefail`，任何命令失败立即中断，避免错误被忽略导致灾难

> ⚠️ 部署前数据库备份是**强制步骤**，禁止用注释或条件跳过，否则更新失败时无法恢复数据

> ⚠️ 备份文件要有清理策略（保留N天），否则磁盘会持续占满最终导致备份失败

> ⚠️ 健康检查超时时间（30×2s=60s）必须大于Spring Boot启动时间，否则正常启动也会被判定为失败触发回滚

> ⚠️ 回滚机制依赖 `.current_tag` 文件记录部署前版本，禁止手动删除该文件

> ⚠️ `docker compose down` 会停止并删除容器，但不会删除命名卷（数据卷），所以数据不会丢失；但 `docker compose down -v` 会删除卷，**禁止在生产环境使用**

> ⚠️ 滚动更新时先更新backend再更新frontend，避免前端请求到旧版backend出现兼容性问题

> ⚠️ SSH远程执行deploy.sh时，环境变量（如DB_USER）可能不会自动加载，要在脚本中source `.env` 或使用 `--env-file` 参数

> ⚠️ 部署日志要定期轮转（logrotate），避免单文件过大难以查看

> ⚠️ 回滚后必须再次健康检查，如果回滚也失败说明问题不是镜像版本问题（可能是配置、数据库schema等），需要人工介入
