## 3. API调用规范

```typescript
// api/sale/order.ts
import apiClient from '@/utils/request';

export interface SaleOrderQuery {
  page: number;
  pageSize: number;
  billNo?: string;
  customerId?: number;
  startDate?: string;
  endDate?: string;
}

export interface SaleOrderListVO {
  id: number;
  billNo: string;
  billStatus: string;
  billDate: string;
  customerName: string;
  totalAmount: number;
}

// 分页查询
export function getSaleOrderPage(params: SaleOrderQuery) {
  return apiClient.get<PageResult<SaleOrderListVO>>('/api/sale/order', { params });
}

// 详情查询
export function getSaleOrderById(id: number) {
  return apiClient.get<SaleOrderDetailVO>(`/api/sale/order/${id}`);
}

// 新增
export function createSaleOrder(data: SaleOrderCreateDTO) {
  return apiClient.post<number>('/api/sale/order', data);
}

// 修改
export function updateSaleOrder(id: number, data: SaleOrderUpdateDTO) {
  return apiClient.put<void>(`/api/sale/order/${id}`, data);
}

// 删除
export function deleteSaleOrder(id: number) {
  return apiClient.delete<void>(`/api/sale/order/${id}`);
}

// 审核
export function auditSaleOrder(id: number, data: AuditDTO) {
  return apiClient.post<void>(`/api/sale/order/${id}/audit`, data);
}
```

**API调用规范要点：**

- API文件按模块拆分，存放在`src/api/{module}/`目录下
- 每个API文件导出TypeScript接口类型（Query/VO/DTO）和请求函数
- 请求函数命名规则：`get{Resource}Page`（分页）、`get{Resource}ById`（详情）、`create{Resource}`（新增）、`update{Resource}`（修改）、`delete{Resource}`（删除）、`audit{Resource}`（审核）
- URL路径与后端RESTful API一一对应：`/api/{module}/{resource}`
- 所有请求/响应必须定义TypeScript类型，**禁止使用any类型**

        <vxe-column field="billStatus" title="状态">

          </template>
        </vxe-column>
        <!-- 更多列 -->
      </vxe-table>

        v-model:current-page="queryParams.page"
        v-model:page-size="queryParams.pageSize"
        :total="total"
        @current-change="loadData"
        @size-change="loadData"
      />
    </el-card>
  </div>
</template>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue';
import { getSaleOrderPage, deleteSaleOrder } from '@/api/sale/order';

// 响应式状态
const loading = ref(false);
const tableData = ref<SaleOrderListVO[]>([]);
const total = ref(0);

// 查询参数
const queryParams = reactive<SaleOrderQuery>({
  page: 1,
  pageSize: 20,
  billNo: '',
});

// 加载数据
const loadData = async () => {
  loading.value = true;
  try {
    const { data } = await getSaleOrderPage(queryParams);
    tableData.value = data.records;
    total.value = data.total;
  } finally {
    loading.value = false;
  }
};

// 初始化
onMounted(() => {
  loadData();
});
</script>

## 5. 路由规范

### 路由结构

采用Vue Router 4.x，所有业务页面统一注册在主布局路由下，共享侧边栏+标签页主布局；路由配置按模块拆分到`router/modules/`目录，由路由实例统一加载：

```
/home                              → 首页P01
/crm/customer                      → 客户中心
/crm/contact                       → 客户联系人
/srm/supplier                      → 供应商中心
/product/center                    → 商品中心
/sale/order                        → 销售订单
/purchase/order                    → 采购订单
/inventory/warehouse               → 仓库定义
/production/notice                 → 生产通知单
/outsource/notice                  → 委外通知单
/receivable/receipt                → 收款单
/payable/payment                   → 付款单
/voucher/voucher                   → 凭证
/asset/card                        → 资产卡片
/approval/definition               → 审批定义
/oa/schedule                       → 日程管理
```

### 路由参数规范

| 参数类型 | 格式 | 示例 | 说明 |
|---------|------|------|------|
| 列表页 | /module/feature | /crm/customer | 列表展示页 |
| 新增 | /module/feature/add | /crm/customer/add | 新增录入页 |
| 编辑 | /module/feature/edit/:id | /crm/customer/edit/123 | 编辑录入页 |
| 详情 | /module/feature/detail/:id | /crm/customer/detail/123 | 详情查看页 |
| 弹窗模式 | 组件内Dialog/Drawer | 不产生新路由 | 弹窗打开 |

### 路由规范要点

- 所有业务页面使用**懒加载**（`() => import('@/views/xxx/index.vue')`）
- 路由守卫统一处理**权限校验**与**动态菜单**加载
- 路由配置按模块拆分到`router/modules/`目录
- 弹窗类操作（新增/编辑/详情）优先使用Dialog/Drawer，不产生新路由
- 仅在全屏独占页面（如大屏P11、画像P12）使用独立路由

| 代码字体 | JetBrains Mono / Menlo | 等宽字体 |
| CSS方案 | Tailwind CSS 4 + SCSS变量 | 原子化布局+Element Plus主题变量 |
| 深色模式 | Element Plus + Tailwind CSS | 一键切换深色/浅色模式 |

### 样式职责划分

- **Element Plus组件**：使用SCSS变量（`variables.scss`）定制主题色
- **Tailwind CSS**：仅用于布局/间距/排版原子类，**不通过Tailwind定义业务主题色**
- **深色模式**：统一通过Element Plus的`dark` class切换，**Tailwind不重复定义`dark:`前缀**
- 全局样式文件结构：`variables.scss`（SCSS变量）+ `element-override.scss`（Element Plus样式覆盖）+ `global.scss`（全局基础样式）
- Tailwind CSS 4入口：`app.css`（CSS-based配置，取代传统`tailwind.config.ts`）

## 7. 状态管理规范

### 全局状态（Pinia）

```typescript
// stores/user.ts
import { defineStore } from 'pinia';
import { getUserInfo, logout } from '@/api/auth';

export const useUserStore = defineStore('user', {
  state: () => ({
    token: '',
    userInfo: null as UserInfo | null,
    permissions: [] as string[],
    roles: [] as string[],
  }),
  
  getters: {
    isLoggedIn: (state) => !!state.token,
    hasPermission: (state) => (permission: string) => state.permissions.includes(permission),
  },
  
  actions: {
    async login(credentials: LoginDTO) {
      const { data } = await loginApi(credentials);
      this.token = data.token;
    },
    
    async fetchUserInfo() {
      const { data } = await getUserInfo();
      this.userInfo = data.user;
      this.permissions = data.permissions;
      this.roles = data.roles;
    },
    
    async logout() {
      await logout();
      this.$reset();
    },
  },
  
  persist: true, // 持久化
});
```

### 状态管理职责划分

| 管理工具 | 职责范围 | 适用场景 |
|---------|---------|---------|
| Pinia（全局状态） | 用户信息、权限、主题、语言、标签页状态 | 全局共享、跨页面持久化数据 |
| Vue Query（服务端状态） | API请求缓存、重试、后台刷新、乐观更新 | 服务端数据的获取与同步 |
| ref/reactive（局部状态） | 页面内临时状态（表单数据、加载状态、弹窗开关） | 组件内部响应式数据 |

### 状态管理规范要点

- Pinia Store使用`defineStore`+选项式API（state/getters/actions）
- Store文件存放在`src/stores/`目录，按功能拆分（auth.ts/menu.ts/tab.ts/ui.ts/permission.ts）
- 需要持久化的Store设置`persist: true`
- 服务端数据优先使用Vue Query管理，避免手动在Pinia中缓存API响应
- 所有状态变量必须定义TypeScript类型，**禁止使用any类型**
