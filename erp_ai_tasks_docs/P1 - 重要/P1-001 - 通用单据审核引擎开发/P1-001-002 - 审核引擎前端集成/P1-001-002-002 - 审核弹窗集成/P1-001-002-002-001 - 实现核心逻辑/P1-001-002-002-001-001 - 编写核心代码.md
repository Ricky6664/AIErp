### 5.1 审核弹窗组件 — AuditDialog.vue

```vue
<!-- erp-web/src/components/business/AuditDialog.vue -->
<template>
  <el-dialog
    v-model="dialogVisible"
    :title="dialogTitle"
    width="500px"
    :close-on-click-modal="false"
    @close="handleClose"
  >
    <el-form ref="formRef" :model="formData" :rules="formRules" label-width="100px">
      <el-form-item label="单据类型"><span>{{ props.docType }}</span></el-form-item>
      <el-form-item label="单据ID"><span>{{ props.docId }}</span></el-form-item>

      <!-- 提交审核：备注可选 -->
      <el-form-item v-if="props.operationType === 'submit'" label="提交备注" prop="opinion">
        <el-input v-model="formData.opinion" type="textarea" :rows="3"
          placeholder="请输入提交备注（可选）" maxlength="500" show-word-limit />
      </el-form-item>

      <!-- 审核通过：意见必填 -->
      <el-form-item v-if="props.operationType === 'approve'" label="审核意见" prop="opinion">
        <el-input v-model="formData.opinion" type="textarea" :rows="3"
          placeholder="请输入审核意见（必填）" maxlength="500" show-word-limit />
      </el-form-item>

      <!-- 审核驳回：原因必填 -->
      <el-form-item v-if="props.operationType === 'reject'" label="驳回原因" prop="opinion">
        <el-input v-model="formData.opinion" type="textarea" :rows="3"
          placeholder="请输入驳回原因（必填）" maxlength="500" show-word-limit />
      </el-form-item>

      <!-- 作废：原因必填 -->
      <el-form-item v-if="props.operationType === 'void'" label="作废原因" prop="opinion">
        <el-input v-model="formData.opinion" type="textarea" :rows="3"
          placeholder="请输入作废原因（必填）" maxlength="500" show-word-limit />
      </el-form-item>

      <!-- 撤销作废：确认提示 -->
      <el-form-item v-if="props.operationType === 'cancelVoid'">
        <el-alert title="确认撤销作废？撤销后单据将恢复至作废前状态。"
          type="warning" :closable="false" />
      </el-form-item>
    </el-form>

    <template #footer>
      <el-button @click="handleClose">取 消</el-button>
      <el-button :type="submitButtonType" :loading="submitLoading"
        @click="handleSubmit">确 定</el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref, computed, watch, reactive } from 'vue'
import { ElMessage, type FormInstance } from 'element-plus'
import { submitAudit, approveAudit, rejectAudit, voidAudit, cancelVoidAudit }
  from '@/api/engine/audit'

interface AuditDialogProps {
  visible: boolean
  operationType: 'submit' | 'approve' | 'reject' | 'void' | 'cancelVoid'
  docType: string
  docId: number
}

const props = defineProps<AuditDialogProps>()
const emit = defineEmits<{
  'update:visible': [value: boolean]
  'success': []
}>()

const dialogVisible = computed({
  get: () => props.visible,
  set: (val) => emit('update:visible', val),
})

const formRef = ref<FormInstance>()
const submitLoading = ref(false)
const formData = reactive({ opinion: '' })

const dialogTitle = computed(() => {
  const titleMap: Record<string, string> = {
    submit: '提交审核', approve: '审核通过', reject: '审核驳回',
    void: '单据作废', cancelVoid: '撤销作废',
  }
  return titleMap[props.operationType] || '审核操作'
})

const submitButtonType = computed(() => {
  const typeMap: Record<string, string> = {
    submit: 'primary', approve: 'success', reject: 'danger',
    void: 'danger', cancelVoid: 'warning',
  }
  return typeMap[props.operationType] || 'primary'
})

const formRules = computed(() => {
  const requiredOps = ['approve', 'reject', 'void']
  if (requiredOps.includes(props.operationType)) {
    return { opinion: [
      { required: true, message: '请输入意见/原因', trigger: 'blur' },
      { max: 500, message: '最多500个字符', trigger: 'blur' },
    ]}
  }
  return { opinion: [{ max: 500, message: '最多500个字符', trigger: 'blur' }] }
})

watch(() => props.visible, (val) => {
  if (val) { formData.opinion = ''; formRef.value?.resetFields() }
})

async function handleSubmit() {
  if (formRef.value) {
    try { await formRef.value.validate() } catch { return }
  }
  submitLoading.value = true
  try {
    const params = { docType: props.docType, docId: props.docId }
    switch (props.operationType) {
      case 'submit': await submitAudit({ ...params, submitRemark: formData.opinion }); break
      case 'approve': await approveAudit({ ...params, opinion: formData.opinion }); break
      case 'reject': await rejectAudit({ ...params, reason: formData.opinion }); break
      case 'void': await voidAudit({ ...params, voidReason: formData.opinion }); break
      case 'cancelVoid': await cancelVoidAudit(params); break
    }
    ElMessage.success(`${dialogTitle.value}操作成功`)
    dialogVisible.value = false
    emit('success')
  } catch (error: any) {
    ElMessage.error(error.message || `${dialogTitle.value}操作失败`)
  } finally {
    submitLoading.value = false
  }
}

function handleClose() { dialogVisible.value = false }
</script>
```

### 5.2 使用示例 — 父组件集成

```vue
<template>
  <div>
    <el-table :data="tableData">
      <el-table-column label="操作">
        <template #default="{ row }">
          <el-button @click="openDialog('approve', row)">审核</el-button>
          <el-button @click="openDialog('void', row)">作废</el-button>
        </template>
      </el-table-column>
    </el-table>
    <AuditDialog v-model:visible="dialogVisible" :operation-type="currentOperation"
      :doc-type="currentDoc.docType" :doc-id="currentDoc.docId" @success="refreshList" />
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import AuditDialog from '@/components/business/AuditDialog.vue'

const dialogVisible = ref(false)
const currentOperation = ref<'submit' | 'approve' | 'reject' | 'void' | 'cancelVoid'>('approve')
const currentDoc = ref({ docType: '', docId: 0 })

function openDialog(op: typeof currentOperation.value, row: any) {
  currentOperation.value = op
  currentDoc.value = { docType: row.docType, docId: row.docId }
  dialogVisible.value = true
}
function refreshList() { fetchListData() }
</script>
```

### 5.3 关键设计要点

1. **v-model:visible**：使用Vue3 defineProps + computed get/set实现弹窗显隐的双向绑定
2. **动态标题**：根据operationType计算弹窗标题和提交按钮类型（primary/success/danger/warning）
3. **条件表单**：不同操作类型显示不同表单字段，approve/reject/void的opinion为必填，submit的submitRemark可选，cancelVoid仅需确认
4. **表单校验**：使用computed动态计算rules，必填操作required=true
5. **事件通知**：操作成功后emit('success')通知父组件刷新列表，取消操作不触发API调用

### 5.4 数据交互

- 涉及API：根据operationType调用对应的审核API端点
- 涉及组件：AuditDialog.vue（通用弹窗）, 父组件（列表页或详情页）

## 八、易错警示

> ⚠️ v-model:visible需使用Vue3 computed get/set模式，不可直接修改props

> ⚠️ 表单校验规则需根据operationType动态计算，不可硬编码

> ⚠️ 弹窗打开时必须重置表单数据（watch visible），避免上次操作残留

> ⚠️ emit('success')仅在API成功时触发，取消和失败不触发

> ⚠️ TypeScript类型定义需完整，operationType使用联合类型约束

> ⚠️ el-dialog的close-on-click-modal=false，防止误点关闭丢失已填内容
