/** 获取单据审核状态及允许操作 */

### 5.2 Composable — useAuditButtons.ts

```typescript
// erp-web/src/composables/engine/useAuditButtons.ts
import { ref, watch, onMounted, type Ref } from 'vue'
import { getAuditStatus, type AuditStatusVO } from '@/api/engine/audit'

/**
 * 审核按钮状态联动 Composable
 *
 * @param docType 单据类型标识（如 'sale_order', 'purchase_order'）
 * @param docId 单据ID（响应式引用，单据切换时自动刷新）
 *
 * @example
 * const { canSubmit, canApprove, loading, refresh } =
 *   useAuditButtons('sale_order', toRef(form, 'id'))
 */
export function useAuditButtons(docType: string, docId: Ref<number | undefined>) {
  const canSubmit = ref(false)
  const canApprove = ref(false)
  const canReject = ref(false)
  const canUnconfirm = ref(false)
  const canVoid = ref(false)
  const canCancelVoid = ref(false)
  const canWithdraw = ref(false)

  const currentStatus = ref<number | null>(null)
  const loading = ref(false)
  const auditStatusData = ref<AuditStatusVO | null>(null)

  async function fetchAuditStatus() {
    if (!docId.value) {
      resetAllFlags()
      return
    }
    loading.value = true
    try {
      const res = await getAuditStatus({ docType, docId: docId.value })
      auditStatusData.value = res.data
      currentStatus.value = res.data.currentStatus

      const ops = res.data.allowedOperations
      canSubmit.value = ops.includes('SUBMIT')
      canApprove.value = ops.includes('APPROVE')
      canReject.value = ops.includes('REJECT')
      canUnconfirm.value = ops.includes('UNAUDIT')
      canVoid.value = ops.includes('VOID')
      canCancelVoid.value = ops.includes('UNVOID')
      canWithdraw.value = ops.includes('WITHDRAW')
    } catch (error) {
      console.error('获取审核状态失败:', error)
      resetAllFlags()
    } finally {
      loading.value = false
    }
  }

  function resetAllFlags() {
    canSubmit.value = false
    canApprove.value = false
    canReject.value = false
    canUnconfirm.value = false
    canVoid.value = false
    canCancelVoid.value = false
    canWithdraw.value = false
    currentStatus.value = null
    auditStatusData.value = null
  }

  onMounted(() => { fetchAuditStatus() })

  watch(docId, () => { fetchAuditStatus() })

  return {
    canSubmit, canApprove, canReject, canUnconfirm,
    canVoid, canCancelVoid, canWithdraw,
    currentStatus, loading, auditStatusData,
    refresh: fetchAuditStatus,
  }
}
```

### 5.3 使用示例 — 业务单据页面

```vue
<template>
  <div class="doc-detail">
    <el-button-group>
      <el-button v-if="canSubmit" v-permission="'audit:submit'"
        type="primary" @click="handleSubmit">提交审核</el-button>
      <el-button v-if="canApprove" v-permission="'audit:approve'"
        type="success" @click="handleApprove">审核通过</el-button>
      <el-button v-if="canReject" v-permission="'audit:reject'"
        type="danger" @click="handleReject">审核驳回</el-button>
      <el-button v-if="canWithdraw" v-permission="'audit:withdraw'"
        @click="handleWithdraw">撤回</el-button>
      <el-button v-if="canUnconfirm" v-permission="'audit:unconfirm'"
        type="warning" @click="handleUnconfirm">反审</el-button>
      <el-button v-if="canVoid" v-permission="'audit:void'"
        type="danger" @click="handleVoid">作废</el-button>
      <el-button v-if="canCancelVoid" v-permission="'audit:cancelVoid'"
        @click="handleCancelVoid">撤销作废</el-button>
    </el-button-group>
  </div>
</template>

<script setup lang="ts">
import { toRef } from 'vue'
import { useAuditButtons } from '@/composables/engine/useAuditButtons'

const form = reactive({ id: 1001 })
const { canSubmit, canApprove, canReject, canUnconfirm,
  canVoid, canCancelVoid, canWithdraw, refresh } =
  useAuditButtons('sale_order', toRef(form, 'id'))

async function handleSubmit() {
  await submitAudit({ docType: 'sale_order', docId: form.id })
  await refresh()
}
</script>
```

### 5.4 关键设计要点

1. **Composable模式**：使用Vue3 Composition API，接收docType和响应式docId参数，返回所有按钮状态标志
2. **自动刷新**：watch监听docId变化，单据切换时自动重新获取审核状态
3. **手动刷新**：暴露refresh方法，审核操作（submit/approve等）完成后调用，刷新按钮状态
4. **loading状态**：请求过程中loading=true，按钮可配合显示加载动画
5. **v-permission指令**：按钮同时使用v-if（状态控制）和v-permission（权限控制），双重保障

### 5.5 数据交互

- 涉及API：GET /api/engine/audit/status（获取审核状态及允许操作）
- 涉及VO：AuditStatusVO(docType, docId, currentStatus, allowedOperations[])

## 八、易错警示

> ⚠️ docId参数必须是Ref类型（使用toRef传递），确保单据切换时watch能触发

> ⚠️ 审核操作（submit/approve等）完成后必须调用refresh()刷新按钮状态，否则按钮不会更新

> ⚠️ v-permission指令的权限标识需与后端@PreAuthorize注解的权限一致

> ⚠️ API请求失败时resetAllFlags，避免显示错误的操作按钮

> ⚠️ TypeScript类型定义需与后端AuditStatusVO保持一致
