### 5.1 辅助方法清单

```java
/** 编码生成（幂等调用） */
private String generateCode() {
    return codeGeneratorService.generate("sale");
}

/** Entity→VO转换 */
private SaleQuotationDetailVO convertToDetailVO(SaleQuotationEntity entity) {
    SaleQuotationDetailVO vo = BeanUtil.copyProperties(entity, SaleQuotationDetailVO.class);
    vo.setDetails(getDetailList(entity.getId()));
    return vo;
}

/** 金额汇总计算 */
private BigDecimal calculateTotal(List<SaleQuotationDetailEntity> details) {
    return details.stream()
        .map(d -> AmountCalculator.calc(d.getPrice(), d.getQty(), d.getTaxRate(), d.getDiscount()))
        .reduce(BigDecimal.ZERO, BigDecimal::add);
}
```
