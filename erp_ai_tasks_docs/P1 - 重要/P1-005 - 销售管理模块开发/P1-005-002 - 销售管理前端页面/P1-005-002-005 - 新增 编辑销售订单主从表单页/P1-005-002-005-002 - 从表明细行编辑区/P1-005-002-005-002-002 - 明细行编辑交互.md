### 5.1 编辑交互逻辑
```typescript
// 新增明细行
const addDetailRow = () => {
  detailTableRef.value.insertAt({ qty: 1, price: 0, amount: 0 }, -1)
}

// 删除明细行
const removeDetailRow = (row: any) => {
  detailTableRef.value.remove(row)
  recalcTotal()
}

// 单元格编辑事件
const onCellEdit = ({ row, column }: any) => {
  if (['qty', 'price'].includes(column.field)) {
    row.amount = Number((row.qty * row.price).toFixed(4))
    recalcTotal()
  }
}
```
