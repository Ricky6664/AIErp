### 5.1 Service接口定义
```java
public interface IProdWorkOrderService extends IServiceX<ProdWorkOrderEntity> {
    /** 分页查询列表 */
    Page<ProdWorkOrderListVO> pageList(ProdWorkOrderQueryDTO query);

    /** 查询详情 */
    ProdWorkOrderDetailVO getDetail(Long id);

    /** 新增 */
    Long create(ProdWorkOrderCreateDTO dto);

    /** 修改 */
    Boolean update(ProdWorkOrderUpdateDTO dto);

    /** 删除 */
    Boolean delete(Long id);

    /** 提交审核 */
    Boolean submitAudit(Long id, String opinion);
}
```
