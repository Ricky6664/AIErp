### 5.1 辅助方法清单

```java
/** 编码生成（幂等调用） */
private String generateCode() {
    return codeGeneratorService.generate("prod");
}

/** Entity→VO转换 */
private ProdWorkOrderDetailVO convertToDetailVO(ProdWorkOrderEntity entity) {
    ProdWorkOrderDetailVO vo = BeanUtil.copyProperties(entity, ProdWorkOrderDetailVO.class);
    vo.setDetails(getDetailList(entity.getId()));
    return vo;
}

/** 金额汇总计算 */
private BigDecimal calculateTotal(List<ProdWorkOrderDetailEntity> details) {
    return details.stream()
        .map(d -> AmountCalculator.calc(d.getPrice(), d.getQty(), d.getTaxRate(), d.getDiscount()))
        .reduce(BigDecimal.ZERO, BigDecimal::add);
}
```
