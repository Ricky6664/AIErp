### 5.1 联动规则
```typescript
// 客户变更→联动刷新
watch(() => formData.customerId, async (newVal) => {
  if (newVal) {
    // 刷新客户价格策略
    priceStrategy.value = await api.getPriceStrategy(newVal)
    // 刷新客户收货地址
    addresses.value = await api.getCustomerAddresses(newVal)
  }
})

// 仓库变更→联动刷新库位列表
watch(() => formData.warehouseId, async (newVal) => {
  if (newVal) {
    locations.value = await api.getStorageLocations(newVal)
  }
})
```
