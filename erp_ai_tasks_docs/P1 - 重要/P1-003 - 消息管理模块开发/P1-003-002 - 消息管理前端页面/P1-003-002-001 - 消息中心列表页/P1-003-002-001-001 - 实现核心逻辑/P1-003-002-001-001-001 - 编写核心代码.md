### 5.2 核心逻辑（script setup）
```typescript
import { ref, onMounted } from 'vue'
import { getMessagePage, markRead, markAllRead as markAllReadApi } from '@/api/msg/message'
import type { MessageListVO } from '@/types/msg'

const activeTab = ref('all')
const keyword = ref('')
const messages = ref<MessageListVO[]>([])
const total = ref(0)
const pageNum = ref(1)
const pageSize = 20
const typeTabs = ref([
  { code: 'all', label: '全部', unreadCount: 0 },
  { code: 'system', label: '系统消息', unreadCount: 0 },
  { code: 'business', label: '业务消息', unreadCount: 0 },
  { code: 'warning', label: '预警消息', unreadCount: 0 },
  { code: 'todo', label: '待办消息', unreadCount: 0 },
])

const loadMessages = async () => {
  pageNum.value = 1
  const res = await getMessagePage({ typeId: activeTab.value, keyword: keyword.value, pageNum: 1, pageSize })
  messages.value = res.data.records
  total.value = res.data.total
}

const loadMore = async () => {
  if (messages.value.length >= total.value) return
  pageNum.value++
  const res = await getMessagePage({ typeId: activeTab.value, keyword: keyword.value, pageNum: pageNum.value, pageSize })
  messages.value.push(...res.data.records)
}

const handleRead = async (msg: MessageListVO) => {
  if (!msg.isRead) { await markRead(msg.id); msg.isRead = true }
  currentMsg.value = msg
  drawerVisible.value = true
}

const markAllReadAction = async () => {
  await markAllReadApi()
  messages.value.forEach(m => m.isRead = true)
  typeTabs.value.forEach(t => t.unreadCount = 0)
}
```

## 八、易错警示

> ⚠️ 无限滚动需正确处理边界条件：数据加载完毕（length >= total）后停止请求，避免重复加载

> ⚠️ 未读数角标需在标记已读/全部已读后同步更新，不能依赖下次轮询

> ⚠️ 消息内容支持HTML格式，使用v-html渲染时需确保后端已做XSS过滤

> ⚠️ Tab切换时需重置分页状态（pageNum=1，messages清空），避免数据错乱
