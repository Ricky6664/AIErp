### 5.2 核心逻辑（script setup）
```typescript
import { ref, onMounted, onUnmounted } from 'vue'
import { useRouter } from 'vue-router'
import { ElNotification } from 'element-plus'
import { Bell } from '@element-plus/icons-vue'
import { getUnreadCount } from '@/api/msg/message'
import { useWebSocket } from '@/composables/useWebSocket'

const router = useRouter()
const unreadCount = ref(0)
let pollTimer: ReturnType<typeof setInterval> | null = null

// WebSocket连接
const { connect, disconnect, onMessage } = useWebSocket('/api/msg/ws/connect')

onMessage((msg: { title: string; content: string; priority: string; type: string }) => {
  unreadCount.value++
  const notifyType = msg.priority === 'urgent' ? 'error' : msg.priority === 'important' ? 'warning' : 'info'
  const duration = msg.priority === 'urgent' ? 0 : 5000  // 紧急常驻，普通5s
  ElNotification({
    title: msg.title,
    message: msg.content,
    type: notifyType as any,
    duration,
    position: 'bottom-right',
    onClick: () => router.push('/msg/center')
  })
})

// 30s轮询兜底
const fetchUnreadCount = async () => {
  try {
    const res = await getUnreadCount()
    unreadCount.value = res.data
  } catch (e) { /* 静默失败 */ }
}

const goToMessageCenter = () => { router.push('/msg/center') }

onMounted(() => {
  connect()
  fetchUnreadCount()
  pollTimer = setInterval(fetchUnreadCount, 30000)
})

onUnmounted(() => {
  disconnect()
  if (pollTimer) clearInterval(pollTimer)
})
```

### 5.3 WebSocket composable封装
```typescript
// composables/useWebSocket.ts
import { ref } from 'vue'

export function useWebSocket(url: string) {
  const ws = ref<WebSocket | null>(null)
  const messageHandlers: Function[] = []
  let reconnectTimer: ReturnType<typeof setTimeout> | null = null

  const connect = () => {
    const token = localStorage.getItem('token')
    ws.value = new WebSocket(`${location.origin.replace('http', 'ws')}${url}?token=${token}`)
    ws.value.onopen = () => { console.log('[WS] connected') }
    ws.value.onmessage = (event) => {
      const data = JSON.parse(event.data)
      if (data.type === 'pong') return  // 心跳响应
      messageHandlers.forEach(fn => fn(data))
    }
    ws.value.onclose = () => { reconnectTimer = setTimeout(connect, 5000) }
    // 30s心跳
    setInterval(() => { ws.value?.send(JSON.stringify({ type: 'ping' })) }, 30000)
  }

  const onMessage = (handler: Function) => { messageHandlers.push(handler) }
  const disconnect = () => { ws.value?.close(); if (reconnectTimer) clearTimeout(reconnectTimer) }

  return { connect, disconnect, onMessage }
}
```

## 八、易错警示

> ⚠️ WebSocket连接URL需正确处理ws/wss协议切换，HTTPS环境下必须使用wss

> ⚠️ 组件onUnmounted必须清理WebSocket连接和setInterval定时器，避免内存泄漏

> ⚠️ 心跳机制（30s ping）是防止代理/负载均衡器超时断开连接的关键，不可省略

> ⚠️ 轮询兜底接口需做好异常静默处理，网络抖动时不应弹出错误提示影响用户
