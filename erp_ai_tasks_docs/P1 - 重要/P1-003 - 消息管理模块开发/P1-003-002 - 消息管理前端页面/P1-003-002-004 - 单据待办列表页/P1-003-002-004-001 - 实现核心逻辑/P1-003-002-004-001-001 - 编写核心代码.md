### 5.2 核心逻辑（script setup）
```typescript
import { ref, reactive, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { getTodoPage, approveTodo, rejectTodo, batchApproveTodo } from '@/api/msg/todo'
import type { TodoListVO } from '@/types/msg'
import { ElMessage, ElMessageBox } from 'element-plus'

const router = useRouter()
const activeTab = ref('all')
const tableData = ref<TodoListVO[]>([])
const total = ref(0)
const pageNum = ref(1)
const pageSize = 20
const selectedIds = ref<number[]>([])
const opinionMap = reactive<Record<number, string>>({})
const countMap = reactive({ pending_approval: 0, pending_handle: 0, pending_confirm: 0 })
const businessTypeMap: Record<string, string> = { sale_order: '销售订单', purchase_order: '采购订单', payment: '付款单', receipt: '收款单' }
const todoTypeLabel: Record<string, string> = { pending_approval: '待审核', pending_handle: '待审批', pending_confirm: '待确认' }
const todoTypeColor: Record<string, string> = { pending_approval: 'warning', pending_handle: 'primary', pending_confirm: 'info' }

const loadData = async () => {
  const res = await getTodoPage({ todoType: activeTab.value === 'all' ? '' : activeTab.value, isCompleted: false, pageNum: pageNum.value, pageSize })
  tableData.value = res.data.records; total.value = res.data.total
}

const doApprove = async (row: TodoListVO) => {
  await approveTodo({ todoId: row.id, opinion: opinionMap[row.id] || '' })
  ElMessage.success('审批成功'); loadData()
}

const batchApprove = async () => {
  await ElMessageBox.confirm(`确认批量审批 ${selectedIds.value.length} 条待办？`, '提示', { type: 'warning' })
  await batchApproveTodo({ todoIds: selectedIds.value, opinion: '批量审批通过' })
  ElMessage.success('批量审批成功'); loadData()
}

const goToDoc = (row: TodoListVO) => { router.push({ path: `/${row.businessType}/${row.businessId}` }) }
```

### 5.3 数据交互
- 调用API：GET /api/message/todo（分页查询）、POST /api/message/todo/{id}/approve（审批）、POST /api/message/todo/{id}/reject（驳回）、POST /api/message/todo/batch-approve（批量审批）
- 路由路径：/msg/todo

> ⚠️ 审批意见Popover在表格滚动时需正确跟随，建议使用ref管理Popover实例
