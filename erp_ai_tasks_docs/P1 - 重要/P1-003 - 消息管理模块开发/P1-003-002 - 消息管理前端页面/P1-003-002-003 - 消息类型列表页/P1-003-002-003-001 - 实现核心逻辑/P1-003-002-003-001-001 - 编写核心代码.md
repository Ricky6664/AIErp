### 5.2 核心逻辑（script setup）
```typescript
import { ref, reactive, onMounted } from 'vue'
import { getTypeList, createType, updateType, deleteType } from '@/api/msg/type'
import type { TypeListVO } from '@/types/msg'
import { ElMessage, ElMessageBox } from 'element-plus'

const treeData = ref<(TypeListVO & { children?: any[] })[]>([])
const currentNode = ref<TypeListVO | null>(null)
const childList = ref<TypeListVO[]>([])
const dialogVisible = ref(false)
const isEdit = ref(false)
const formRef = ref()
const formData = reactive({ id: undefined as number | undefined, parentId: undefined as number | undefined, typeCode: '', typeName: '', sortNo: 0, icon: '', enableFlag: true })

const buildTree = (list: TypeListVO[]): (TypeListVO & { children?: any[] })[] => {
  const map = new Map<number, TypeListVO & { children?: any[] }>()
  const roots: (TypeListVO & { children?: any[] })[] = []
  list.forEach(item => map.set(item.id, { ...item, children: [] }))
  list.forEach(item => {
    const node = map.get(item.id)!
    if (item.parentId && map.has(item.parentId)) { map.get(item.parentId)!.children!.push(node) }
    else { roots.push(node) }
  })
  return roots
}

const loadTree = async () => {
  const res = await getTypeList()
  treeData.value = buildTree(res.data)
}

const onNodeClick = (data: TypeListVO & { children?: TypeListVO[] }) => {
  currentNode.value = data
  childList.value = data.children || []
}
```

## 八、易错警示

> ⚠️ 树数据需前端从扁平列表构建，注意parentId为0或null时作为根节点处理

> ⚠️ el-tree-select的check-strictly属性必须开启，否则选中子节点会自动勾选父节点导致parentId错误

> ⚠️ 删除类型前需后端校验是否有关联的msg_message记录，前端需展示后端返回的具体错误提示

> ⚠️ 树节点展开/折叠状态在CRUD操作后需保持，避免操作后整棵树收起影响用户体验
