### 5.2 核心逻辑（script setup）
```typescript
import { ref, reactive, onMounted, onUnmounted, nextTick } from 'vue'
import * as echarts from 'echarts'
import { getWarningDashboard, handleWarning as handleWarningApi } from '@/api/msg/warning'

const trendChartRef = ref<HTMLElement>()
const pieChartRef = ref<HTMLElement>()
let trendChart: echarts.ECharts | null = null
let pieChart: echarts.ECharts | null = null
let refreshTimer: ReturnType<typeof setInterval> | null = null
const activeModule = ref('')
const warningList = ref<any[]>([])

const kpiCards = reactive([
  { module: 'sale', label: '销售预警', count: 0, color: '#E6A23C', icon: 'Sell' },
  { module: 'purchase', label: '采购预警', count: 0, color: '#409EFF', icon: 'ShoppingCart' },
  { module: 'inventory', label: '库存预警', count: 0, color: '#F56C6C', icon: 'Box' },
  { module: 'finance', label: '财务预警', count: 0, color: '#67C23A', icon: 'Money' },
])

const moduleLabelMap: Record<string, string> = { sale: '销售', purchase: '采购', inventory: '库存', finance: '财务' }
const warningTypeLabel: Record<string, string> = { inventory: '库存', receivable: '应收', payable: '应付', delivery: '交期' }
const severityLabel: Record<string, string> = { info: '提示', warning: '警告', urgent: '紧急' }
const severityColor: Record<string, string> = { info: 'info', warning: 'warning', urgent: 'danger' }

const loadDashboard = async () => {
  const res = await getWarningDashboard({ module: activeModule.value })
  // 更新KPI卡片
  kpiCards.forEach(card => { card.count = res.data.moduleCounts[card.module] || 0 })
  // 更新预警明细
  warningList.value = res.data.warningList
  // 更新图表
  renderTrendChart(res.data.trendData)
  renderPieChart(res.data.distributionData)
}

const renderTrendChart = (data: { dates: string[]; counts: number[] }) => {
  if (!trendChart) trendChart = echarts.init(trendChartRef.value!)
  trendChart.setOption({
    xAxis: { type: 'category', data: data.dates },
    yAxis: { type: 'value' },
    series: [{ data: data.counts, type: 'line', smooth: true, areaStyle: { opacity: 0.3 } }],
    tooltip: { trigger: 'axis' }
  })
}

const renderPieChart = (data: { name: string; value: number }[]) => {
  if (!pieChart) pieChart = echarts.init(pieChartRef.value!)
  pieChart.setOption({
    series: [{ type: 'pie', radius: ['40%', '70%'], data }],
    tooltip: { trigger: 'item' },
    legend: { bottom: 0 }
  })
}

const onCardClick = (module: string) => { activeModule.value = activeModule.value === module ? '' : module; loadDashboard() }

onMounted(() => { loadDashboard(); refreshTimer = setInterval(loadDashboard, 60000) })
onUnmounted(() => { if (refreshTimer) clearInterval(refreshTimer); trendChart?.dispose(); pieChart?.dispose() })
```

## 八、易错警示

> ⚠️ ECharts实例必须在onMounted后、DOM渲染完成后初始化（nextTick），否则会报容器未找到错误

> ⚠️ 组件onUnmounted时必须dispose ECharts实例和clearInterval定时器，避免内存泄漏

> ⚠️ 窗口resize时需调用chart.resize()保持图表自适应，建议监听window resize事件

> ⚠️ 60s自动刷新时如果用户正在操作表格（如点击处理），需避免刷新覆盖操作状态
