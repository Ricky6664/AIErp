### 5.1 格式化方法
```typescript
// 日期格式化
const formatDate = ({ cellValue }: any) => {
  return cellValue ? dayjs(cellValue).format('YYYY-MM-DD HH:mm:ss') : '-'
}

// 金额格式化（保留4位小数）
const formatAmount = ({ cellValue }: any) => {
  return cellValue != null ? Number(cellValue).toFixed(4) : '-'
}
```
